function(rru_test name)
    add_executable(${name} ${name}.c)
    target_link_libraries(${name} PRIVATE rru_core)
    target_compile_definitions(${name} PRIVATE _GNU_SOURCE)
    target_compile_options(${name} PRIVATE -Wall -Wextra)
    add_test(NAME ${name} COMMAND ${name} ${ARGN})
endfunction()

rru_test(test_sx1255)
rru_test(test_ring)
rru_test(test_latency)
set_tests_properties(test_latency PROPERTIES TIMEOUT 60)

set_tests_properties(test_e2e PROPERTIES TIMEOUT 180)
